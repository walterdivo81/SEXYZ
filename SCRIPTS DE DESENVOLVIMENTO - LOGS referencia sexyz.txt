export function calcularIdade(nascimento: string | null | undefined, hoje = new Date()): number | null {
  if (!nascimento) return null;
  const d = new Date(nascimento + "T00:00:00");
  if (isNaN(d.getTime())) return null;
  let idade = hoje.getFullYear() - d.getFullYear();
  const m = hoje.getMonth() - d.getMonth();
  if (m < 0 || (m === 0 && hoje.getDate() < d.getDate())) idade--;
  return idade;
}

export function maiorDeIdade(nascimento: string, hoje = new Date()) {
  const i = calcularIdade(nascimento, hoje);
  return i !== null && i >= 18;
}

import { useQuery } from "@tanstack/react-query";
import { supabase } from "@/integrations/supabase/client";

/** Converte caminhos guardados no perfil em links temporários (o armazenamento é privado). */
export function useFotoUrls(paths: string[] | null | undefined) {
  const list = (paths ?? []).filter(Boolean);
  return useQuery({
    queryKey: ["fotos", list],
    enabled: list.length > 0,
    staleTime: 50 * 60 * 1000,
    queryFn: async () => {
      const { data, error } = await supabase.storage.from("fotos").createSignedUrls(list, 3600);
      if (error) throw error;
      const map: Record<string, string> = {};
      data?.forEach((d) => { if (d.path && d.signedUrl) map[d.path] = d.signedUrl; });
      return map;
    },
  });
}

import { useEffect, useState } from "react";
import type { User } from "@supabase/supabase-js";
import { supabase } from "@/integrations/supabase/client";

export function useUser() {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);
  useEffect(() => {
    supabase.auth.getUser().then(({ data }) => { setUser(data.user); setLoading(false); });
    const { data } = supabase.auth.onAuthStateChange((_e, s) => setUser(s?.user ?? null));
    return () => data.subscription.unsubscribe();
  }, []);
  return { user, loading };
}

import { useQuery } from "@tanstack/react-query";
import { supabase } from "@/integrations/supabase/client";
import type { Database } from "@/integrations/supabase/types";

export type Perfil = Database["public"]["Tables"]["profiles"]["Row"];

/** Curtidas recebidas e enviadas, com os perfis envolvidos, e a lista de matches (curtida mútua). */
export function useCoSEXYZes(userId: string) {
  return useQuery({
    queryKey: ["coSEXYZes", userId],
    queryFn: async () => {
      const { data: likes, error } = await supabase
        .from("likes")
        .select("liker_id, liked_id, created_at")
        .or(`liker_id.eq.${userId},liked_id.eq.${userId}`)
        .order("created_at", { ascending: false });
      if (error) throw error;
      const enviei = new Set(likes.filter((l) => l.liker_id === userId).map((l) => l.liked_id));
      const recebi = new Set(likes.filter((l) => l.liked_id === userId).map((l) => l.liker_id));
      const ids = [...new Set([...enviei, ...recebi])];
      let perfis: Perfil[] = [];
      if (ids.length) {
        const { data, error: e2 } = await supabase.from("profiles").select("*").in("id", ids);
        if (e2) throw e2;
        perfis = data;
      }
      const byId = new Map(perfis.map((p) => [p.id, p]));
      const pick = (s: string[]) => s.map((id) => byId.get(id)).filter(Boolean) as Perfil[];
      return {
        curtiramVoce: pick([...recebi].filter((id) => !enviei.has(id))),
        vocêCurtiu: pick([...enviei].filter((id) => !recebi.has(id))),
        matches: pick([...enviei].filter((id) => recebi.has(id))),
      };
    },
  });
}

    queryKey: ["descobrir", filtros],

              {aba === "matches" && <Button asChild className="w-full"><Link to="/mensagens" search={{ com: p.id }}><MessageCircle className="size-4" /> Conversar</Link></Button>}
              {aba === "vocêCurtiu" && <Button variant="outline" className="w-full" onClick={() => descurtir(p)}>Desfazer curtida</Button>}
            </PerfilCard>
          ))}
        </div>
      )}
    </div>
  );
}

src/routes/_authenticated/perfil.tsx


import { createFileRoute } from "@tanstack/react-router";
import { useQuery, useQueryClient } from "@tanstack/react-query";
import { useEffect, useRef, useState } from "react";
import { toast } from "sonner";
import { Plus, Star, Trash2 } from "lucide-react";
import { supabase } from "@/integrations/supabase/client";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Textarea } from "@/components/ui/textarea";
import { selectClass } from "@/components/AuthCard";
import { useFotoUrls } from "@/lib/fotos";
import { ESTADOS, GENEROS, MAX_FOTOS } from "@/lib/regras";

export const Route = createFileRoute("/_authenticated/perfil")({
  head: () => ({
    meta: [
      { title: "Meu perfil | SEXYZ" },
      { name: "description", content: "Edite suas informações e fotos." },
      { property: "og:title", content: "Meu perfil | SEXYZ" },
      { property: "og:description", content: "Edite suas informações e fotos." },
    ],
  }),
  component: Perfil,
});

function Perfil() {
  const { userId } = Route.useRouteContext();
  const qc = useQueryClient();
  const fileRef = useRef<HTMLInputElement>(null);
  const [enviando, setEnviando] = useState(false);
  const { data: perfil } = useQuery({
    queryKey: ["perfil", userId],
    queryFn: async () => {
      const { data, error } = await supabase.from("profiles").select("*").eq("id", userId).single();
      if (error) throw error;
      return data;
    },
  });
  const [f, setF] = useState({ nome: "", bio: "", cidade: "", estado: "", interesses: "", genero: "" });
  useEffect(() => {
    if (perfil) setF({ nome: perfil.nome, bio: perfil.bio ?? "", cidade: perfil.cidade ?? "", estado: perfil.estado ?? "", interesses: perfil.interesses ?? "", genero: perfil.genero ?? "" });
  }, [perfil]);
  const fotos = perfil?.fotos ?? [];
  const { data: urls = {} } = useFotoUrls(fotos);

  async function salvarFotos(novas: string[]) {
    const { error } = await supabase.from("profiles").update({ fotos: novas }).eq("id", userId);
    if (error) throw error;
    qc.invalidateQueries({ queryKey: ["perfil", userId] });
  }

  async function adicionar(files: FileList | null) {
    if (!files?.length) return;
    const livres = MAX_FOTOS - fotos.length;
    const lista = Array.from(files).slice(0, livres);
    if (!lista.length) return toast.error(`Máximo de ${MAX_FOTOS} fotos.`);
    setEnviando(true);
    try {
      const novos: string[] = [];
      for (const file of lista) {
        if (!file.type.startsWith("image/")) continue;
        if (file.size > 5 * 1024 * 1024) { toast.error(`${file.name} passa de 5 MB.`); continue; }
        const ext = file.name.split(".").pop() || "jpg";
        const path = `${userId}/${Date.now()}-${Math.random().toString(36).slice(2, 7)}.${ext}`;
        const { error } = await supabase.storage.from("fotos").upload(path, file, { contentType: file.type });
        if (error) { toast.error("Falha ao enviar " + file.name); continue; }
        novos.push(path);
      }
      if (novos.length) { await salvarFotos([...fotos, ...novos]); toast.success("Fotos adicionadas!"); }
    } finally {
      setEnviando(false);
      if (fileRef.current) fileRef.current.value = "";
    }
  }

  async function remover(path: string) {
    await supabase.storage.from("fotos").remove([path]);
    await salvarFotos(fotos.filter((p) => p !== path));
  }
  async function principal(path: string) {
    await salvarFotos([path, ...fotos.filter((p) => p !== path)]);
  }

  async function salvar() {
    const { error } = await supabase.from("profiles").update(f).eq("id", userId);
    if (error) return toast.error("Erro ao salvar.");
    toast.success("Perfil atualizado!");
    qc.invalidateQueries({ queryKey: ["perfil", userId] });
  }

  const set = (k: keyof typeof f) => (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement | HTMLSelectElement>) => setF({ ...f, [k]: e.target.value });

  return (
    <div>
      <div className="mb-6 flex items-end justify-between">
        <div><p className="text-sm font-semibold uppercase tracking-widest text-accent">Seu espaço</p><h1 className="text-4xl font-extrabold">Meu perfil</h1></div>
        <Button onClick={salvar}>Salvar alterações</Button>
      </div>
      <div className="grid gap-6 lg:grid-cols-2">
        <section className="space-y-4 rounded-3xl border bg-card p-6">
          <h3 className="text-xl font-bold">Informações pessoais</h3>
          <div className="space-y-1.5"><Label>Apelido</Label><Input value={f.nome} onChange={set("nome")} /></div>
          <div className="space-y-1.5"><Label>Biografia</Label><Textarea rows={4} value={f.bio} onChange={set("bio")} placeholder="Conte um pouco sobre você" /></div>
          <div className="space-y-1.5"><Label>Gênero</Label>
            <select className={selectClass} value={f.genero} onChange={set("genero")}><option value="">Selecione</option>{GENEROS.map((g) => <option key={g}>{g}</option>)}</select></div>
          <div className="grid grid-cols-[1fr_6rem] gap-3">
            <div className="space-y-1.5"><Label>Cidade</Label><Input value={f.cidade} onChange={set("cidade")} /></div>
            <div className="space-y-1.5"><Label>UF</Label>
              <select className={selectClass} value={f.estado} onChange={set("estado")}><option value="">--</option>{ESTADOS.map((u) => <option key={u}>{u}</option>)}</select></div>
          </div>
          <div className="space-y-1.5"><Label>Interesses (separados por vírgula)</Label><Input value={f.interesses} onChange={set("interesses")} placeholder="Música, viagens, cinema" /></div>
        </section>

        <section className="rounded-3xl border bg-card p-6">
          <h3 className="text-xl font-bold">Galeria de fotos</h3>
          <p className="mb-4 text-sm text-muted-foreground">Até {MAX_FOTOS} fotos. A primeira é a principal. ({fotos.length}/{MAX_FOTOS})</p>
          <div className="grid grid-cols-3 gap-3">
            {Array.from({ length: MAX_FOTOS }).map((_, i) => {
              const path = fotos[i];
              return path ? (
                <div key={path} className="group relative aspect-square overflow-hidden rounded-2xl bg-secondary">
                  {urls[path] && <img src={urls[path]} alt={`Foto ${i + 1}`} className="h-full w-full object-cover" />}
                  {i === 0 && <span className="absolute left-2 top-2 rounded-full bg-primary px-2 py-0.5 text-[10px] font-bold text-primary-foreground">Principal</span>}
                  <div className="absolute inset-x-0 bottom-0 flex justify-center gap-1 bg-background/70 p-1 opacity-0 transition-opacity group-hover:opacity-100">
                    {i !== 0 && <Button size="icon" variant="ghost" className="size-7" aria-label="Tornar principal" onClick={() => principal(path)}><Star className="size-4" /></Button>}
                    <Button size="icon" variant="ghost" className="size-7" aria-label="Remover foto" onClick={() => remover(path)}><Trash2 className="size-4" /></Button>
                  </div>
                </div>
              ) : (
                <button key={i} disabled={enviando} onClick={() => fileRef.current?.click()}
                  className="grid aspect-square place-items-center rounded-2xl border-2 border-dashed text-muted-foreground transition-colors hover:border-primary hover:text-primary">
                  <span className="flex flex-col items-center text-xs"><Plus className="size-5" />Foto {i + 1}</span>
                </button>
              );
            })}
          </div>
          <input ref={fileRef} type="file" accept="image/*" multiple hidden onChange={(e) => adicionar(e.target.files)} />
          <Button className="mt-4 w-full" variant="secondary" disabled={enviando || fotos.length >= MAX_FOTOS} onClick={() => fileRef.current?.click()}>
            {enviando ? "Enviando..." : "Adicionar fotografias"}
          </Button>
        </section>
      </div>
    </div>
  );
}

export const Route = createFileRoute("/_authenticated/configuracoes")({
  head: () => ({
    meta: [
      { title: "Configurações | SEXYZ" },
      { name: "description", content: "Senha e conta." },
      { property: "og:title", content: "Configurações | SEXYZ" },
      { property: "og:description", content: "Senha e conta." },
    ],
  }),
  component: Config,
});

function Config() {
  const nav = useNavigate();
  const [atual, setAtual] = useState("");
  const [nova, setNova] = useState("");

  async function trocar(e: React.FormEvent) {
    e.preventDefault();
    if (nova.length < 6) return toast.error("A nova senha precisa ter pelo menos 6 caracteres.");
    const { error } = await supabase.auth.updateUser({ password: nova, current_password: atual } as never);
    if (error) return toast.error("Não foi possível alterar. Confira a senha atual.");
    toast.success("Senha alterada!");
    setAtual(""); setNova("");
  }

  async function sair() {
    await supabase.auth.signOut();
    nav({ to: "/login" });
  }

  return (
    <div className="max-w-lg space-y-6">
      <h1 className="text-4xl font-extrabold">Configurações</h1>
      <form onSubmit={trocar} className="space-y-4 rounded-3xl border bg-card p-6">
        <h3 className="text-xl font-bold">Alterar senha</h3>
        <div className="space-y-1.5"><Label>Senha atual</Label><Input type="password" value={atual} onChange={(e) => setAtual(e.target.value)} /></div>
        <div className="space-y-1.5"><Label>Nova senha</Label><Input type="password" value={nova} onChange={(e) => setNova(e.target.value)} /></div>
        <Button type="submit">Salvar nova senha</Button>
      </form>
      <div className="rounded-3xl border bg-card p-6">
        <h3 className="mb-3 text-xl font-bold">Sessão</h3>
        <Button variant="destructive" onClick={sair}>Sair da conta</Button>
      </div>
    </div>
  );
}
